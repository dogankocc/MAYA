Status BpeTokenizer::Train(const std::string& corpus, std::size_t targetVocabSize) {
  // 1. Corpus'u kelimelere ayir
  std::vector<Word> words = SplitCorpusIntoWords(corpus);
  
  // 2. Baslangic sembollerini vocab'a ekle
  CollectSymbols(words, vocabulary_);

  // 3. Hedef vocab boyutuna ulasana kadar merge yap
  while (vocabulary_.Size() < targetVocabSize) {
    // En sik gorulen cifti bul
    const auto frequencies = CountPairFrequencies(words);
    const auto best = std::max_element(frequencies.begin(), frequencies.end(),
        [](const auto& lhs, const auto& rhs) { return lhs.second < rhs.second; });

    // Merge islemi
    const std::string left = ...;
    const std::string right = ...;
    const std::string merged = left + right;

    // Kelimelerde merge uygula
    for (Word& word : words) {
      MergePairInWord(word, left, right, merged);
    }

    vocabulary_.AddToken(merged);
    merges_.push_back({left, right});
    mergeRank_[MergeKey(left, right)] = merges_.size() - 1;
  }
  return Status::Ok();
}

Encoding Islemi

std::vector<TokenId> BpeTokenizer::Encode(const std::string& text) const {
  std::vector<TokenId> ids;
  std::istringstream stream(text);
  std::string word;

  while (stream >> word) {
    std::vector<std::string> symbols;
    symbols.push_back(kWordStartToken);  // _ karakteri
    
    // Karakter karakter ayir
    for (const char ch : word) {
      symbols.emplace_back(1, ch);
    }

    // BPE kurallarini uygula
    const std::vector<std::string> tokens = ApplyBpe(symbols);
    
    // Token'leri ID'ye cevir
    for (const std::string& token : tokens) {
      ids.push_back(vocabulary_.GetId(token));
    }
  }
  return ids;
}

Decoding Islemi

std::string BpeTokenizer::Decode(const std::vector<TokenId>& ids) const {
  std::string text;
  for (const TokenId id : ids) {
    // Ozel tokenleri atla
    if (id == kPadTokenId || id == kBosTokenId || id == kEosTokenId) {
      continue;
    }

    const std::string& token = vocabulary_.GetToken(id);
    
    // WordStart (_) ise bosluk ekle
    if (token == kWordStartToken) {
      if (!text.empty()) text.push_back(' ');
      continue;
    }
    text += token;
  }
  return text;
}

3.1 Linear Layer Forward

Lineer katman: output = input * W^T (veya quantize edilmis W)

// llm/src/model/linear.cpp
Status Linear::Forward(const Tensor& input, Tensor& output) const {
  // input: [seq_len, input_dim]
  // output: [seq_len, output_dim]
  
  if (useQuantizedWeight_) {
    // Quantize edilmis agirlik ile matmul
    return tensor::MatMulQuantized(input, quantizedWeight_, output);
  }

  // Normal FP32 matmul: W^T ile carp
  const Tensor weightTransposed = weight_.Transpose2D();
  const Status matmulStatus = tensor::MatMul(input, weightTransposed, output);
  
  // Bias ekle (varsa)
  if (useBias_) {
    const Dimension seqLen = input.GetShape()[0];
    for (Dimension row = 0; row < seqLen; ++row) {
      for (Dimension col = 0; col < outputDim_; ++col) {
        output.At({row, col}) += bias_[col];
      }
    }
  }
  return Status::Ok();
}

3.2 Multi-Head Attention Mekanizmasi

Multi-Head Attention (MHA), birden fazla attention kafasi kullanarak baglanti kurar.

Attention Formulu

Attention(Q, K, V) = softmax(Q*K^T / sqrt(d_k)) * V

Q: Query (sorgu) - Ben ne ariyorum?
K: Key (anahtar) - Ben ne sunuyorum?
V: Value (deger) - Ben ne veriyorum?

Multi-Head: Q, K, V'yi birden fazla kafa boyunca bol,
her kafa icin ayri attention hesapla, sonuclari birlestir.

C++ Implementasyonu

// llm/src/model/attention.cpp
Status MultiHeadAttention::Forward(const Tensor& input, const RopeCache& rope, Tensor& output) const {
  const Dimension seqLen = input.GetShape()[0];
  const Dimension numHeads = config_.numHeads;
  const Dimension headDim = config_.hiddenDim / numHeads;
  const Scalar scale = 1.0f / std::sqrt(static_cast<Scalar>(headDim));

  // 1. Q, K, V projection'larini uygula
  Tensor queries, keys, values;
  query_.Forward(input, queries);   // Q = input * W_q
  key_.Forward(input, keys);        // K = input * W_k
  value_.Forward(input, values);    // V = input * W_v

  // 2. Rotary Position Embedding (RoPE) uygula
  ApplyRope(queries, rope, numHeads, headDim);
  ApplyRope(keys, rope, numKvHeads, headDim);

  // 3. Her kafa icin attention hesapla
  Tensor merged = Tensor::Zeros(Shape{seqLen, config_.hiddenDim});
  
  for (Dimension queryPos = 0; queryPos < seqLen; ++queryPos) {
    for (Dimension head = 0; head < numHeads; ++head) {
      // a. Attention skorlarini hesapla: Q*K^T / sqrt(d_k)
      std::vector<Scalar> scores(seqLen, -INFINITY);
      for (Dimension keyPos = 0; keyPos <= queryPos; ++keyPos) {  // Causal mask
        Scalar dot = 0.0f;
        for (Dimension dim = 0; dim < headDim; ++dim) {
          Scalar q = queries.At({queryPos, head * headDim + dim});
          Scalar k = keys.At({keyPos, kvHead * headDim + dim});
          dot += q * k;
        }
        scores[keyPos] = dot * scale;
      }

      // b. Softmax uygula (numerik kararlilik icin max cikar)
      Scalar maxScore = *max_element(scores.begin(), scores.begin() + queryPos + 1);
      Scalar sumExp = 0.0f;
      for (Dimension keyPos = 0; keyPos <= queryPos; ++keyPos) {
        scores[keyPos] = std::exp(scores[keyPos] - maxScore);
        sumExp += scores[keyPos];
      }

      // c. Agirlikli toplam: sum(attention_weight * V)
      for (Dimension dim = 0; dim < headDim; ++dim) {
        Scalar weighted = 0.0f;
        for (Dimension keyPos = 0; keyPos <= queryPos; ++keyPos) {
          Scalar attnWeight = scores[keyPos] / sumExp;
          Scalar v = values.At({keyPos, kvHead * headDim + dim});
          weighted += attnWeight * v;
        }
        merged.At({queryPos, head * headDim + dim}) = weighted;
      }
    }
  }

  // 4. Output projection: merged * W_o
  return output_.Forward(merged, output);
}

3.3 SwiGLU FFN

SwiGLU (Swish-Gated Linear Unit), LLaMA'da kullanilan modern FFN varyantidir.

Standart FFN vs SwiGLU

TipFormulParametre
Standart FFNFFN(x) = max(0, x*W1 + b1) * W2 + b22 Linear
SwiGLU FFNFFN(x) = (SiLU(x*Wg) * (x*Wu)) * Wd3 Linear (gate, up, down)

SiLU (Swish) Aktivasyonu

SiLU(x) = x * sigmoid(x)
sigmoid(x) = 1 / (1 + exp(-x))

SiLU turevi: sigmoid(x) * (1 + x * (1 - sigmoid(x)))

C++ Implementasyonu

// llm/src/model/ffn.cpp
Status SwiGluFfn::Forward(const Tensor& input, Tensor& output) const {
  Tensor gate, up, hidden;

  // 1. Gate ve Up projection'lari
  gate_.Forward(input, gate);   // gate = x * W_g
  up_.Forward(input, up);       // up = x * W_u

  // 2. SiLU(gate) * up uygula
  tensor::Silu(gate);           // gate = SiLU(gate)
  tensor::Mul(gate, up, hidden);  // hidden = gate * up

  // 3. Down projection
  return down_.Forward(hidden, output);  // output = hidden * W_d
}

// llm/src/tensor/ops.cpp - SiLU implementasyonu
void Silu(Tensor& tensor) {
  for (Index i = 0; i < static_cast<Index>(tensor.Numel()); ++i) {
    const Scalar x = tensor[i];
    const Scalar sigmoid = 1.0f / (1.0f + std::exp(-x));
    tensor[i] = x * sigmoid;
  }
}

3.4 Transformer Block (Pre-LN, Residual)

Her Transformer katmani iki ana kisimdan olusur: Attention + FFN. Pre-LN ve residual baglantilar kullanilir.

Pre-LN vs Post-LN

Pre-LN (LLaMA, MAYA):
x1 = x + Attention(RMSNorm(x))
x2 = x1 + FFN(RMSNorm(x1))

Post-LN (Orijinal Transformer):
x1 = RMSNorm(x + Attention(x))
x2 = RMSNorm(x1 + FFN(x1))

Pre-LN avantaji: Egitim daha stabil, gradientler daha iyi akar.

RMSNorm Formulu

RMS = sqrt(mean(x^2))
RMSNorm(x, weight) = weight * (x / (RMS + eps))

RMS, x karelerinin ortalamasının kareköküdür.
LayerNorm'dan farki: mean cikarmaz, sadece olcekleme yapar.

C++ Implementasyonu

// llm/src/model/transformer_block.cpp
Status TransformerBlock::Forward(const Tensor& input, const RopeCache& rope, Tensor& output) const {
  Tensor attnNormed, attnOutput, residual;
  Tensor ffnNormed, ffnOutput;

  // === Attention Sub-Layer ===
  // 1. RMSNorm uygula (Pre-LN)
  RmsNorm(input, attnNormWeight_, config_.normEps, attnNormed);
  
  // 2. Multi-Head Attention
  attention_.Forward(attnNormed, rope, attnOutput);
  
  // 3. Residual baglanti: x + Attention(RMSNorm(x))
  tensor::Add(input, attnOutput, residual);

  // === FFN Sub-Layer ===
  // 4. RMSNorm uygula
  RmsNorm(residual, ffnNormWeight_, config_.normEps, ffnNormed);
  
  // 5. SwiGLU FFN
  ffn_.Forward(ffnNormed, ffnOutput);
  
  // 6. Residual baglanti
  return tensor::Add(residual, ffnOutput, output);
}

// llm/src/model/rms_norm.hpp
Status RmsNorm(const Tensor& input, const Tensor& weight, const Scalar eps, Tensor& output) {
  const Dimension seqLen = input.GetShape()[0];
  const Dimension hiddenDim = input.GetShape()[1];

  output = Tensor::Zeros(input.GetShape());

  for (Dimension row = 0; row < seqLen; ++row) {
    // 1. Karelerin toplamini hesapla
    Scalar sumSquares = 0.0f;
    for (Dimension dim = 0; dim < hiddenDim; ++dim) {
      const Scalar value = input.At({row, dim});
      sumSquares += value * value;
    }

    // 2. RMS hesapla
    const Scalar rms = std::sqrt(sumSquares / static_cast<Scalar>(hiddenDim) + eps);
    const Scalar invRms = 1.0f / rms;

    // 3. Normalize et ve olcekle
    for (Dimension dim = 0; dim < hiddenDim; ++dim) {
      const Scalar x = input.At({row, dim});
      const Scalar w = weight[dim];
      output.At({row, dim}) = w * x * invRms;
    }
  }
  return Status::Ok();
}

3.5 Transformer Model (Genel Bakis)

// llm/src/model/transformer.cpp
Status TransformerModel::Forward(const std::vector<TokenId>& tokens, Tensor& logits) const {
  const Dimension seqLen = tokens.size();

  // 1. Token Embedding: token -> hidden state
  Tensor hidden = Tensor::Zeros(Shape{seqLen, config_.hiddenDim});
  for (Dimension pos = 0; pos < seqLen; ++pos) {
    const TokenId token = tokens[pos];
    for (Dimension dim = 0; dim < config_.hiddenDim; ++dim) {
      hidden.At({pos, dim}) = tokenEmbedding_.At({token, dim});
    }
  }

  // 2. Transformer katmanlarini uygula
  Tensor current = hidden;
  for (const TransformerBlock& layer : layers_) {
    Tensor next = Tensor::Zeros(Shape{seqLen, config_.hiddenDim});
    layer.Forward(current, rope_, next);
    current = next;
  }

  // 3. Final RMSNorm
  Tensor normalized;
  RmsNorm(current, finalNormWeight_, config_.normEps, normalized);

  // 4. LM Head: logits = normalized * W_lm
  return LmHeadForward(normalized, logits);
}

4. EGITIM SURECI (Training)

Egitim sureci: Forward Pass -> Loss Hesaplama -> Backward Pass -> Optimizer Step

4.1 Forward Pass Detaylari

Forward Pass, girdi tokenlerinin modele verilerek logitlerin hesaplanmasidir.

Akis:
tokens -> token_embedding -> [N transformer katmani] -> RMSNorm -> LM_Head -> logits

logits: [seq_len, vocab_size] - Her token icin vocab olasiliklari

4.2 CrossEntropy Loss Hesaplama

CrossEntropy, siniflandirma problemlerinde kullanilan standart loss fonksiyonudur.

Formul

Softmax: p_i = exp(logit_i - max_logit) / sum(exp(logit_j - max_logit))

CrossEntropy: loss = -sum(target_one_hot * log(p)) / N

MAYA'da:
- logits'ten softmax ile olasilik hesapla
- target token'in olasiligini bul
- -log(prob) ile loss hesapla
- Gradient: p_i - target_i (hedef sinif icin 1, digerleri icin 0)

C++ Implementasyonu

// llm/src/training/loss.cpp
float CrossEntropyLoss(const Tensor& logits, const std::vector<TokenId>& targetTokens, Tensor& gradLogits) {
  const Dimension seqLen = logits.GetShape()[0];
  const Dimension vocabSize = logits.GetShape()[1];

  gradLogits = Tensor::Zeros(logits.GetShape());
  float totalLoss = 0.0f;

  for (Dimension row = 0; row < seqLen; ++row) {
    const TokenId target = targetTokens[row];

    // 1. Numerik kararlilik icin max_logit cikar
    Scalar maxLogit = logits.At({row, 0});
    for (Dimension vocab = 1; vocab < vocabSize; ++vocab) {
      maxLogit = std::max(maxLogit, logits.At({row, vocab}));
    }

    // 2. Softmax olasiliklarini hesapla
    Scalar sumExp = 0.0f;
    for (Dimension vocab = 0; vocab < vocabSize; ++vocab) {
      const Scalar probability =
          std::exp(logits.At({row, vocab}) - maxLogit);
      gradLogits.At({row, vocab}) = probability;  // Ara deger
      sumExp += probability;
    }

    // 3. Normalize et
    for (Dimension vocab = 0; vocab < vocabSize; ++vocab) {
      gradLogits.At({row, vocab}) /= sumExp;  // Artik p_i
    }

    // 4. Loss hesapla: -log(p_target)
    const Scalar targetProbability = gradLogits.At({row, target});
    totalLoss -= std::log(std::max(targetProbability, 1e-12f));

    // 5. Gradient hesapla: d_loss/d_logit = p_i - target_i
    // target_i: hedef sinif icin 1, digerleri icin 0
    gradLogits.At({row, target}) -= 1.0f;  // p_target - 1
    
    // Ortalama al (batch/seq normalization)
    for (Dimension vocab = 0; vocab < vocabSize; ++vocab) {
      gradLogits.At({row, vocab}) /= static_cast<Scalar>(seqLen);
    }
  }

  return totalLoss / static_cast<Scalar>(seqLen);
}

4.3 Backward Pass (Gradient Hesaplama, Chain Rule)

Backward Pass, kayiptan (loss) geriye dogru gradient hesaplamadir. Zincir kurali (Chain Rule) kullanilir.

Zincir Kurali (Chain Rule)

Eger y = f(g(x)) ise:
dy/dx = dy/dg * dg/dx

Ornek: L = CrossEntropy(Softmax(Linear(Transformer(x))))

dL/dW_lm_head = dL/dLogits * dLogits/dW_lm_head
dL/dW_attn = ... ve boyle devam eder

MatMul Backward

// Eger output = a * b (matmul) ise
// d_output/d_a = b^T
// d_output/d_b = a^T

// llm/src/training/autograd.cpp
Status MatMulBackward(const Tensor& a, const Tensor& b, const Tensor& gradOutput,
                      Tensor& gradA, Tensor& gradB) {
  // gradA = gradOutput * b^T
  // gradB = a^T * gradOutput
  
  const Tensor bTransposed = b.Transpose2D();
  const Tensor aTransposed = a.Transpose2D();

  // dL/da = dL/dOutput * b^T
  tensor::MatMul(gradOutput, bTransposed, gradA);
  
  // dL/db = a^T * dL/dOutput
  tensor::MatMul(aTransposed, gradOutput, gradB);
  
  return Status::Ok();
}

Linear Backward

// Linear: y = x * W^T
// dL/dx = dL/dy * W
// dL/dW = dL/dy^T * x

Status LinearBackward(const Tensor& input, const Tensor& weight, const Tensor& gradOutput,
                      Tensor& gradInput, Tensor& gradWeight) {
  const Tensor weightTransposed = weight.Transpose2D();
  return MatMulBackward(input, weightTransposed, gradOutput, gradInput, gradWeight);
}

SiLU Backward

// SiLU(x) = x * sigmoid(x)
// dSiLU/dx = sigmoid(x) * (1 + x * (1 - sigmoid(x)))

void SiluBackward(const Tensor& input, const Tensor& output, const Tensor& gradOutput,
                  Tensor& gradInput) {
  gradInput = Tensor::Zeros(input.GetShape());

  for (Index i = 0; i < static_cast<Index>(input.Numel()); ++i) {
    const Scalar x = input[i];
    const Scalar sigmoid = 1.0f / (1.0f + std::exp(-x));
    const Scalar derivative = sigmoid * (1.0f + x * (1.0f - sigmoid));
    gradInput[i] = gradOutput[i] * derivative;
  }
}

4.4 AdamW Optimizer Calisma Prensibi

AdamW, Adam optimizerinin weight decay duzeltilmis halidir. Modern LLM egitiminde en cok kullanilan optimizerdir.

Adam Formulleri

m_t = beta1 * m_{t-1} + (1 - beta1) * g_t      (Birinci moment - ortalama)
v_t = beta2 * v_{t-1} + (1 - beta2) * g_t^2    (Ikinci moment - varyans)

m_hat = m_t / (1 - beta1^t)                      (Bias duzeltme)
v_hat = v_t / (1 - beta2^t)                      (Bias duzeltme)

theta_t = theta_{t-1} - lr * m_hat / (sqrt(v_hat) + eps)

AdamW'da ayrica weight decay:
theta_t = theta_t - lr * weight_decay * theta_{t-1}

Hiperparametreler

ParametreTipik DegerAciklama
learning_rate1e-3 veya 3e-4Ogrenme orani
beta10.9Birinci moment ustel ortalama
beta20.999Ikinci moment ustel ortalama
epsilon1e-8Sifira bolunmeyi onle
weight_decay0.01Agirlik ayristirma (L2 regularizasyon)

C++ Implementasyonu

// llm/src/training/adamw.cpp
Status AdamW::Step(ParameterList& parameters) {
  // Ilk adimda moment tensorlerini baslat
  if (moment1_.size() != parameters.Size()) {
    moment1_.clear();
    moment2_.clear();
    for (const Parameter& p : parameters.Parameters()) {
      moment1_.push_back(Tensor::Zeros(p.tensor->GetShape()));
      moment2_.push_back(Tensor::Zeros(p.tensor->GetShape()));
    }
  }

  ++step_;
  
  // Bias duzeltme faktörleri
  const float biasCorrection1 = 1.0f - std::pow(config_.beta1, static_cast<float>(step_));
  const float biasCorrection2 = 1.0f - std::pow(config_.beta2, static_cast<float>(step_));
  const float stepSize = config_.learningRate * std::sqrt(biasCorrection2) / biasCorrection1;

  // Her parametre icin guncelleme
  for (std::size_t index = 0; index < parameters.Size(); ++index) {
    Parameter& parameter = parameters.Parameters()[index];
    Tensor& moment1 = moment1_[index];
    Tensor& moment2 = moment2_[index];

    for (Index element = 0; element < static_cast<Index>(parameter.tensor->Numel()); ++element) {
      const Scalar gradient = parameter.grad[element];

      // 1. Birinci moment (m_t)
      moment1[element] = config_.beta1 * moment1[element] + 
                         (1.0f - config_.beta1) * gradient;

      // 2. Ikinci moment (v_t)
      moment2[element] = config_.beta2 * moment2[element] + 
                         (1.0f - config_.beta2) * gradient * gradient;

      // 3. Bias duzeltilmis degerler
      const Scalar mHat = moment1[element] / biasCorrection1;
      const Scalar vHat = moment2[element] / biasCorrection2;

      // 4. Parametre guncelleme
      const Scalar update = stepSize * mHat / (std::sqrt(vHat) + config_.epsilon);
      (*parameter.tensor)[element] -= update;

      // 5. Weight Decay (ayri uygulanir - AdamW ozelligi)
      (*parameter.tensor)[element] -= config_.learningRate * config_.weightDecay * 
                                       (*parameter.tensor)[element];
    }
  }

  return Status::Ok();
}

    WriteTensor(output, prefix + ".attn_norm_weight", block.AttnNormWeight());
    WriteTensor(output, prefix + ".ffn_norm_weight", block.FfnNormWeight());
    WriteTensor(output, prefix + ".attn.q_weight", block.AttentionModule().QueryProjection().Weight());
    WriteTensor(output, prefix + ".attn.k_weight", block.AttentionModule().KeyProjection().Weight());
    WriteTensor(output, prefix + ".attn.v_weight", block.AttentionModule().ValueProjection().Weight());
    WriteTensor(output, prefix + ".attn.o_weight", block.AttentionModule().OutputProjection().Weight());
    WriteTensor(output, prefix + ".ffn.gate_weight", block.FfnModule().GateProjection().Weight());

5. QUANTIZATION (INT8 SIKISTIRMA)

Quantization, model agirliklarini daha az bitli sayilarla temsil etme teknigidir. Bellek kullanimi azalir, hiz artar.

5.1 FP32 -> INT8 Donusumu

MAYA'da symmetric quantization kullanilir. Sıfır simetrik olarak ölçeklenir.

Formul

FP32 -> INT8:
scale = max_abs(FP32_degerleri) / 127.0
INT8_deger = round(FP32_deger / scale)
clamp: -127 ile +127 arasinda

INT8 -> FP32 (De-quantization):
FP32_deger = INT8_deger * scale

Neden 127?

INT8 (signed) -128 ile +127 arasindadir.
Simetrik quantization'da -127 ile +127 kullanilir.
Boylece 0 noktasi tam ortaya gelir.

C++ Implementasyonu

// llm/include/llm/quantization/quantize.hpp
struct QuantizedTensor {
  Shape shape;
  float scale = 1.0f;           // Olcekleme faktoru
  std::vector<std::int8_t> data;  // INT8 veriler
};

// llm/src/quantization/quantize.cpp
QuantizedTensor QuantizeSymmetric(const Tensor& tensor) {
  QuantizedTensor quantized;
  quantized.shape = tensor.GetShape();
  quantized.data.resize(tensor.Numel());

  // 1. Mutlak degerlerin maksimumunu bul
  float maxAbs = 0.0f;
  for (Index i = 0; i < static_cast<Index>(tensor.Numel()); ++i) {
    maxAbs = std::max(maxAbs, std::fabs(tensor[i]));
  }

  // Sıfır tensörü ise
  if (maxAbs < 1e-8f) {
    quantized.scale = 1.0f;
    std::fill(quantized.data.begin(), quantized.data.end(), 
              static_cast<std::int8_t>(0));
    return quantized;
  }

  // 2. Scale hesapla
  quantized.scale = maxAbs / 127.0f;

  // 3. Her elemani INT8'e cevir
  for (Index i = 0; i < static_cast<Index>(tensor.Numel()); ++i) {
    const float scaled = tensor[i] / quantized.scale;
    const auto rounded = static_cast<long>(std::lround(scaled));
    // Clamp: -127 ile +127 arasinda
    const auto clamped = std::clamp(rounded, -127L, 127L);
    quantized.data[static_cast<std::size_t>(i)] = 
        static_cast<std::int8_t>(clamped);
  }

  return quantized;
}

// De-quantization
Tensor Dequantize(const QuantizedTensor& quantized) {
  std::vector<Scalar> data(quantized.data.size());
  for (std::size_t i = 0; i < quantized.data.size(); ++i) {
    data[i] = static_cast<Scalar>(quantized.data[i]) * quantized.scale;
  }
  return Tensor::FromBuffer(quantized.shape, std::move(data));
}

5.2 Quantize Edilmis Linear Layer

Linear layer, quantized agirliklarla direkt matmul yapabilir.

// llm/src/model/linear.cpp
Status Linear::Forward(const Tensor& input, Tensor& output) const {
  if (useQuantizedWeight_) {
    // Quantized agirlik ile hizli matmul
    return tensor::MatMulQuantized(input, quantizedWeight_, output);
  }
  // Normal FP32 matmul
  const Tensor weightTransposed = weight_.Transpose2D();
  return tensor::MatMul(input, weightTransposed, output);
}

6. CIKARIM SURECI (Inference)

Cıkarım, eğitilmiş modelin yeni tokenler üretmesidir. İki ana aşama: Prefill ve Decode.

6.1 Prefill Aşaması

Prefill, kullanıcının girdi (prompt) tokenlerinin modelden geçirilmesidir.

Özellikleri

- Tüm prompt tokenleri aynı anda işlenir (paralel)
- KV Cache doldurulur
- Genellikle daha yavaş (çok token)
- Prompt uzunluğu kadar compute gerekir

C++ Implementasyonu

// llm/src/inference/inference_engine.cpp
Status InferenceEngine::Prefill(const std::vector<TokenId>& tokens, Tensor& logits) {
  Reset();  // Cache'i sıfırla

  // 1. Token embedding
  Tensor hidden;
  EmbedTokens(tokens, hidden);

  // 2. Forward ile tüm tokenleri isle (cacheStart = 0)
  Tensor output;
  ForwardHidden(hidden, 0, output);

  // 3. Son tokenin logitlerini al
  Tensor lastRow = Tensor::Zeros(Shape{1, config_.hiddenDim});
  const Dimension lastIndex = tokens.size() - 1;
  for (Dimension dim = 0; dim < config_.hiddenDim; ++dim) {
    lastRow.At({0, dim}) = output.At({lastIndex, dim});
  }

  return LogitsFromRow(lastRow, logits);
}

6.2 Decode Aşaması

Decode, her adımda bir token üretme ve KV Cache'i güncelleme işlemidir.

Özellikleri

- Her adımda SADECE 1 token işlenir
- KV Cache kullanılır (Önceden hesaplanan K,V değerleri)
- Çok daha hızlı (her adım az compute)
- Yeni K,V değerleri cache'e eklenir

C++ Implementasyonu

Status InferenceEngine::Decode(TokenId token, Tensor& logits) {
  // 1. Tek token embedding
  Tensor hidden;
  EmbedToken(token, hidden);

  // 2. Cache uzunlugundan basla
  const std::size_t cacheStart = cache_.Length();
  
  // 3. Forward (sadece yeni token, cache kullan)
  Tensor output;
  ForwardHidden(hidden, cacheStart, output);

  // 4. Logitleri don
  return LogitsFromRow(output, logits);
}

6.3 KV Cache Mekanizması

KV Cache, her katmandaki Key ve Value tensörlerini kaydederek tekrarlanan hesaplamaları önler.

Neden KV Cache?

KV Cache olmadan:
Adım 1: [t1, t2, t3] -> tüm attention hesapla
Adım 2: [t1, t2, t3, t4] -> TEKRADAN tüm attention hesapla

KV Cache ile:
Adım 1: [t1, t2, t3] -> K1, V1, K2, V2, K3, V3 CACHE'e kaydet
Adım 2: [t4] -> sadece K4, V4 hesapla, CACHE'den K1-3, V1-3 al

C++ Implementasyonu

// llm/include/llm/inference/kv_cache.hpp
class KvCache {
public:
  explicit KvCache(const ModelConfig& config);
  
  // Yazma: Yeni K,V degerlerini ekle
  Status Write(std::size_t layer, std::size_t position, 
               const Tensor& keys, const Tensor& values);
  
  // Okuma: Kaydedilmis K,V degerlerini al
  const Tensor& Keys(std::size_t layer) const;
  const Tensor& Values(std::size_t layer) const;
  
  std::size_t Length() const { return length_; }

private:
  ModelConfig config_;
  Dimension kvDim_;
  std::size_t length_ = 0;
  std::vector<Tensor> keys_;    // Her katman icin [max_seq, kv_dim]
  std::vector<Tensor> values_;  // Her katman icin [max_seq, kv_dim]
};

// llm/src/inference/kv_cache.cpp
KvCache::KvCache(const ModelConfig& config) : config_(config) {
  const Dimension headDim = config.hiddenDim / config.numHeads;
  kvDim_ = config.numKvHeads * headDim;

  // Her katman icin K ve V tensorlerini ayir
  for (std::size_t layer = 0; layer < config.numLayers; ++layer) {
    keys_.push_back(Tensor::Zeros(Shape{config.maxSeqLen, kvDim_}));
    values_.push_back(Tensor::Zeros(Shape{config.maxSeqLen, kvDim_}));
  }
}

Status KvCache::Write(std::size_t layer, std::size_t position,
                      const Tensor& keys, const Tensor& values) {
  const Dimension seqLen = keys.GetShape()[0];
  
  // Yeni K,V degerlerini dogru pozisyona yaz
  for (Dimension row = 0; row < seqLen; ++row) {
    for (Dimension col = 0; col < kvDim_; ++col) {
      keys_[layer].At({position + row, col}) = keys.At({row, col});
      values_[layer].At({position + row, col}) = values.At({row, col});
    }
  }
  return Status::Ok();
}

ToolResult CalculatorToolExecute(const std::map<std::string, std::string>& args) {
  ToolResult result;
  result.toolName = "calculator";
  
  const std::string& expr = args.at("expression");
  // Islemciyi bul (+, -, *, /)
  char op = 0;
  std::size_t opPos = expr.find_first_of("+-*/");
  
  // Sol ve sag sayilari ayristir
  double left = std::stod(expr.substr(0, opPos));
  double right = std::stod(expr.substr(opPos + 1));
  
  // Islemi yap
  switch (op) {
    case '+': result.content = std::to_string(left + right); break;
    case '-': result.content = std::to_string(left - right); break;
    case '*': result.content = std::to_string(left * right); break;
    case '/': result.content = std::to_string(left / right); break;
  }
  result.success = true;
  return result;
}

7.4 Agent Execute Akisi

// llm/src/agent/local_agent_backend.cpp
AgentResponse LocalAgentBackend::Execute(const AgentRequest& request) {
  AgentResponse response;
  
  const std::size_t maxSteps = std::min(request.maxSteps, config_.maxSteps);
  std::vector<AgentStep> steps;

  for (std::size_t step = 0; step < maxSteps; ++step) {
    AgentStep currentStep;
    currentStep.stepNumber = step + 1;

    // 1. Prompt'u olustur (tools + conversation)
    const std::string prompt = BuildAgentPrompt(
        request.userMessage,
        request.systemPrompt,
        conversation_);

    // 2. Modeli calistir
    const auto result = session_.Complete(prompt, request.maxTokensPerStep, rng_);
    const std::string modelOutput = result.Value();

    // 3. Tool call olup olmadigini kontrol et
    const auto toolCalls = ParseToolCalls(modelOutput);

    if (toolCalls && !toolCalls->empty() && config_.enableTools) {
      // === TOOL CALL ===
      currentStep.thought.action = "tool_call";
      currentStep.thought.toolCalls = *toolCalls;

      // 4. Her tool'u calistir
      std::vector<ToolResult> toolResults;
      for (const ToolCall& call : *toolCalls) {
        const ToolResult toolResult = toolRegistry_.Execute(
            call.toolName, call.arguments);
        toolResults.push_back(toolResult);
      }

      currentStep.toolResults = toolResults;
      currentStep.observation = FormatToolResults(toolResults);

      // 5. Konusmaya ekle
      conversation_.push_back({"assistant", modelOutput});
      conversation_.push_back({"tool", currentStep.observation});
    } else {
      // === FINAL ANSWER ===
      currentStep.thought.action = "final_answer";
      currentStep.thought.reasoning = modelOutput;
      response.finalAnswer = modelOutput;
      response.success = true;
      break;
    }

    steps.push_back(std::move(currentStep));
  }

  response.steps = std::move(steps);
  response.totalSteps = steps.size();
  return response;
}

// llm/src/server/chat_http_server.cpp
Status ChatHttpServer::Run(const std::string& host, const int port) {
  httplib::Server http;
  stopRequested_ = false;

  // CORS basliklari
  http.set_default_headers({
    {"Access-Control-Allow-Origin", "*"},
    {"Access-Control-Allow-Methods", "GET, POST, OPTIONS"},
    {"Access-Control-Allow-Headers", "Content-Type"}
  });

  // OPTIONS preflight
  http.Options(R"(/.*)", [](const httplib::Request&, httplib::Response& res) {
    res.status = 204;
  });

  // Endpointler
  http.Get("/api/v1/health", ...);
  http.Get("/api/v1/config", ...);
  http.Post("/api/v1/config", ...);
  http.Post("/api/v1/reset", ...);
  http.Post("/api/v1/chat", ...);

  // Portu dinle
  if (!http.bind_to_port(host.c_str(), port)) {
    return Status::Fail(ErrorCode::Internal, "failed to bind HTTP port");
  }

  // Arka planda dinlemeye basla
  std::thread serverThread([&http]() { http.listen_after_bind(); });

  // Durana kadar bekle
  while (!stopRequested_) {
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
  }

  // Sunucuyu durdur
  http.stop();
  serverThread.join();

  return Status::Ok();
}
